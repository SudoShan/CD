expression
    : expression '+' expression
{
    if (!fold($$, $1, '+', $3) &&
        !simplify($$, $1, '+', $3))
    {
        sprintf($$, "t%d", ++temp);
        printf("%s = %s + %s\n", $$, $1, $3);
    }
}

| expression '*' expression
{
    if (!fold($$, $1, '*', $3) &&
        !simplify($$, $1, '*', $3))
    {
        sprintf($$, "t%d", ++temp);
        printf("%s = %s * %s\n", $$, $1, $3);
    }
};

| expression '*' expression
{
    sprintf($$, "t%d", ++temp);

    if (strcmp($1, "2") == 0)
        printf("%s = %s + %s\n", $$, $3, $3);

    else if (strcmp($3, "2") == 0)
        printf("%s = %s + %s\n", $$, $1, $1);

    else
        printf("%s = %s * %s\n", $$, $1, $3);
}
